class Solution {
  public boolean isValidSudoku(char[][] board) {
    Set<String> set = new HashSet<>();
    for (int i = 0; i < 9; i++) {
      for (int j = 0; j < 9; j++) {
        char ch = board[i][j];
        // Ignore empty cells
        if (ch == '.') {
          continue;
        }
        int boxIndex = (i / 3) * 3 + (j / 3);
        String row = ch + " in row " + i;
        String col = ch + " in col " + j;
        String box = ch + " in box " + boxIndex;
        // Check duplicates
        if (!set.add(row) ||
            !set.add(col) ||
            !set.add(box)) {
          return false;
        }
      }
    }
    return true;
  }
}
