import java.util.Arrays;
class Solution {
  public int findMinArrowShots(int[][] balloons) {
    if (balloons == null || balloons.length == 0) {
      return 0;
    }
    // Sort by ending coordinate
    Arrays.sort(balloons, (a, b) ->
      Integer.compare(a[1], b[1])
    );
    int arrows = 1;
    int arrowPosition = balloons[0][1];
    for (int i = 1; i < balloons.length; i++) {
      // Current arrow cannot burst this balloon
      if (balloons[i][0] > arrowPosition) {
        arrows++;
        arrowPosition = balloons[i][1];
      }
    }
    return arrows;
  }
}
